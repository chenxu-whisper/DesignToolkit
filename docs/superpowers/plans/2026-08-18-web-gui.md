```python
from pydantic import BaseModel, Field, model_validator


class TaskKind(str, Enum):
    VIDEO_FRAMES = "video_frames"
    MEDIA_RESIZE = "media_resize"


class TaskStatus(str, Enum):
    IDLE = "idle"
    QUEUED = "queued"
    RUNNING = "running"
    SUCCEEDED = "succeeded"
    FAILED = "failed"


class LogLevel(str, Enum):
    INFO = "info"
    PROGRESS = "progress"
    SUCCESS = "success"
    ERROR = "error"


class LogPhase(str, Enum):
    CREATED = "created"
    VALIDATING = "validating"
    PREPARING = "preparing"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"


class VideoFramesTaskRequest(BaseModel):
    input_path: str = Field(min_length=1)
    output_dir: str | None = None
    image_format: Literal["png", "jpg", "jpeg"] = "png"
    size: str | None = None
    fps: float | None = None
    frames: int | None = None
    overwrite: bool = False

    @model_validator(mode="after")
    def validate_sampling(self) -> VideoFramesTaskRequest:
        if self.fps is not None and self.frames is not None:
            raise ValueError("fps and frames cannot both be set")
        return self


class MediaResizeTaskRequest(BaseModel):
    input_path: str = Field(min_length=1)
    output_path: str | None = None
    size: str = Field(min_length=1)


class TaskCreateResponse(BaseModel):
    task_id: str
    status: TaskStatus


class TaskResponse(BaseModel):
    task_id: str

    status: TaskStatus

    started_at: datetime | None = None
```

```python
from __future__ import annotations

from collections import defaultdict
from datetime import UTC, datetime
from queue import Queue
from threading import Lock

from designtoolkit.web.schemas import LogEventResponse, LogLevel, LogPhase


class LogStore:
    def __init__(self) -> None:
        self._events: dict[str, list[LogEventResponse]] = defaultdict(list)
        self._subscribers: dict[str, list[Queue[LogEventResponse]]] = defaultdict(list)
        self._lock = Lock()

    def append(
        self,
        *,
        task_id: str,
        level: LogLevel,
        phase: LogPhase,
        message: str,
        details: dict[str, object] | None = None,
    ) -> LogEventResponse:
        event = LogEventResponse(
            task_id=task_id,
            timestamp=datetime.now(UTC),
            level=level,
            phase=phase,
            message=message,
            details=dict(details or {}),
        )
        with self._lock:
            self._events[task_id].append(event)
            subscribers = list(self._subscribers[task_id])
        for subscriber in subscribers:
            subscriber.put(event)
        return event

    def history(self, task_id: str) -> list[LogEventResponse]:
        with self._lock:
            return list(self._events.get(task_id, []))

    def subscribe(self, task_id: str) -> Queue[LogEventResponse]:
        subscriber: Queue[LogEventResponse] = Queue()
        with self._lock:
            self._subscribers[task_id].append(subscriber)
        return subscriber

    def unsubscribe(
        self,
        task_id: str,
        subscriber: Queue[LogEventResponse],
    ) -> None:
        with self._lock:
            subscribers = self._subscribers.get(task_id, [])
            if subscriber in subscribers:
                subscribers.remove(subscriber)
```

```python
class TaskManager:
    def __init__(
        self,
        *,
        log_store: LogStore,
        exporter: Exporter | None = export_video_frames,
        media_resizer: MediaResizer | None = resize_media,
        backend_factory: BackendFactory | None = None,
        run_async: bool = True,
    ) -> None:
        self.log_store = log_store
        self._exporter = exporter
        self._media_resizer = media_resizer
        self._backend_factory = backend_factory or self._default_backend
        self._run_async = run_async
        self._records: dict[str, TaskRecord] = {}
        self._active_task_id: str | None = None
        self._lock = Lock()

    def reserve_task(
        self,
        *,
        kind: TaskKind,
        input_path: str,
        output_path: str | None,
    ) -> TaskRecord:
        with self._lock:
            if self._active_task_id is not None:
                raise TaskConflictError("A task is already running")
            task = TaskRecord(
                task_id=str(uuid4()),
                kind=kind,
                status=TaskStatus.QUEUED,
                created_at=datetime.now(UTC),
                started_at=None,
                finished_at=None,
                input_path=input_path,
                output_path=output_path,
                error=None,
            )
            self._records[task.task_id] = task
            self._active_task_id = task.task_id
        self.log_store.append(
            task_id=task.task_id,
            level=LogLevel.INFO,
            phase=LogPhase.CREATED,
            message=f"Created {kind.value} task",
            details={"input_path": input_path, "output_path": output_path},
        )
        return task

    def get_task(self, task_id: str) -> TaskRecord | None:
        with self._lock:
            return self._records.get(task_id)

    def mark_finished(
        self,
        task_id: str,
        *,
        status: TaskStatus,
        output_path: str | None = None,
        error: str | None = None,
```

```python
from fastapi import APIRouter, Request

class ApiError(Exception):
    def __init__(self, code: str, message: str, status_code: int) -> None:
        super().__init__(message)
        self.code = code
        self.message = message
        self.status_code = status_code


def error_response(exc: ApiError) -> JSONResponse:
    return JSONResponse(
        status_code=exc.status_code,
        content=ErrorResponse(
            error={"code": exc.code, "message": exc.message}
        ).model_dump(mode="json"),
    )


def create_router(task_manager: TaskManager) -> APIRouter:
    router = APIRouter(prefix="/api")

    @router.get("/health")
    def health() -> dict[str, str]:
        return {"status": "ok"}

    @router.post("/tasks/video-frames", response_model=TaskCreateResponse)
    def create_video_frames(request: VideoFramesTaskRequest) -> TaskCreateResponse:
        try:
            task = task_manager.create_video_frames_task(request)
        except TaskConflictError as exc:
            raise ApiError("task_conflict", str(exc), 409) from exc
        return TaskCreateResponse(task_id=task.task_id, status=task.status)

    @router.post("/tasks/media-resize", response_model=TaskCreateResponse)
    def create_media_resize(request: MediaResizeTaskRequest) -> TaskCreateResponse:
        try:
            task = task_manager.create_media_resize_task(request)
        except TaskConflictError as exc:
            raise ApiError("task_conflict", str(exc), 409) from exc
        return TaskCreateResponse(task_id=task.task_id, status=task.status)

    @router.get("/tasks/{task_id}", response_model=TaskResponse)
    def get_task(task_id: str) -> TaskResponse:
        task = task_manager.get_task(task_id)
        if task is None:
            raise ApiError("task_not_found", f"Task not found: {task_id}", 404)
        return task.to_response()

    return router
```

```python
    @router.get("/tasks/{task_id}/events")
    def stream_task_events(task_id: str) -> StreamingResponse:
        task = task_manager.get_task(task_id)
        if task is None:
            raise ApiError("task_not_found", f"Task not found: {task_id}", 404)

        def event_stream() -> Iterator[str]:
            for event in task_manager.log_store.history(task_id):
                yield _sse_line(event)
            latest = task_manager.get_task(task_id)
            if latest is not None and latest.status in {
                TaskStatus.SUCCEEDED,
                TaskStatus.FAILED,
            }:
                return
            subscriber = task_manager.log_store.subscribe(task_id)
            try:
                while True:
                    event = subscriber.get()
                    yield _sse_line(event)
                    latest = task_manager.get_task(task_id)
                    if latest is not None and latest.status in {
                        TaskStatus.SUCCEEDED,
                        TaskStatus.FAILED,
                    }:
                        return
            finally:
                task_manager.log_store.unsubscribe(task_id, subscriber)

        return StreamingResponse(event_stream(), media_type="text/event-stream")
```

```ts
export type VideoFramesValues = {
  inputPath: string
  outputDir: string
  imageFormat: 'png' | 'jpg' | 'jpeg'
  size: string
  samplingMode: 'original' | 'fps' | 'frames'
  fps: string
  frames: string
  overwrite: boolean
}

export type VideoFramesErrors = Partial<Record<keyof VideoFramesValues, string>>

export function validateVideoFramesValues(values: VideoFramesValues): VideoFramesErrors {
  const errors: VideoFramesErrors = {}
  if (!values.inputPath.trim()) {
    errors.inputPath = '请输入视频路径'
  }
  if (values.size.trim() && !SIZE_PATTERN.test(values.size.trim())) {
    errors.size = '尺寸格式应为 WIDTHxHEIGHT'
  }
  if (values.samplingMode === 'fps' && (!values.fps.trim() || Number(values.fps) <= 0)) {
    errors.fps = 'FPS 必须大于 0'
  }
  if (values.samplingMode === 'frames' && (!values.frames.trim() || Number(values.frames) <= 0)) {
    errors.frames = '总帧数必须大于 0'
  }
  return errors
}
```

```ts
export type VideoFramesPayload = {

  output_path?: string

async function request<T>(path: string, payload: unknown): Promise<T> {
  const response = await fetch(path, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload),
  })
  const body = (await response.json()) as ApiErrorResponse | T
  if (!response.ok) {
    throw new Error((body as ApiErrorResponse).error?.message ?? `Request failed: ${response.status}`)
  }
  return body as T
}

export function createVideoFramesTask(payload: VideoFramesPayload): Promise<TaskCreateResponse> {
  return request<TaskCreateResponse>('/api/tasks/video-frames', payload)
}

export function createMediaResizeTask(payload: MediaResizePayload): Promise<TaskCreateResponse> {
  return request<TaskCreateResponse>('/api/tasks/media-resize', payload)
```

```ts
import type { TaskEvent } from './taskTypes'

export function subscribeTaskEvents(
  taskId: string,
  onEvent: (event: TaskEvent) => void,
  onError: (error: Event) => void,
): () => void {
  const source = new EventSource(`/api/tasks/${taskId}/events`)
  source.onmessage = (message) => {
    onEvent(JSON.parse(message.data) as TaskEvent)
  }
  source.onerror = (error) => {
    onError(error)
    source.close()
  }
  return () => source.close()
}
```
